cmake_minimum_required(VERSION 3.20)
project(typelayout_marshal_fixture VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(typelayout_marshal SHARED library.cpp)
target_compile_definitions(typelayout_marshal PRIVATE TYPELAYOUT_MARSHAL_BUILD)
if(WIN32)
    target_sources(typelayout_marshal PRIVATE exports.def)
endif()
set_target_properties(typelayout_marshal PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_executable(typelayout-marshal-native snapshot.cpp)
target_compile_definitions(typelayout-marshal-native PRIVATE TYPELAYOUT_MARSHAL_CONFIGURATION="$<CONFIG>")
add_executable(marshal-layout-oracle oracle.cpp)
target_link_libraries(marshal-layout-oracle PRIVATE typelayout_marshal)
if(MSVC)
    foreach(target typelayout_marshal typelayout-marshal-native marshal-layout-oracle)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME marshal-layout-oracle COMMAND marshal-layout-oracle)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME marshal-native-snapshot
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_snapshot.py" "$<TARGET_FILE:typelayout-marshal-native>" "$<CONFIG>")
endif()
